SELECT r.id
FROM Demo\Model\Robot r
WHERE r.name LIKE :pattern:
    AND r.type NOT LIKE '%old%'
    AND r.code ILIKE ?1
    AND r.year BETWEEN 1970 AND 2000
    AND r.year NOT BETWEEN 1990 AND 1995
    AND r.status IN (1, 2, ?2)
    AND r.owner NOT IN (SELECT o.id FROM Demo\Model\Owner o)
    AND EXISTS (SELECT p.id FROM Demo\Model\Part p)
    AND r.tags @> '["a"]'
    AND r.meta <@ r.other
    AND r.ranges && r.window
    AND r.document @@ 'search'
    AND r.name || '!' = r.title
    AND r.meta -> 'a' ->> 'b' = r.json #> 'c' #>> 'd'
    AND r.flags & 4 | 2 ^ 1 = ~r.mask
    AND NOT r.deleted
    AND r.price * 2 / 3 % 4 + 1 - 1 >= 100
    AND r.parent IS NOT NULL
