SELECT DISTINCT
    u.*,
    u.name AS title,
    COUNT(p.id) total,
    CASE u.status WHEN 1 THEN 'active' WHEN 2 THEN 'banned' ELSE 'unknown' END,
    CAST(u.rank AS integer),
    CONVERT(u.name USING utf8),
    u.rank * 2 + 1,
    'literal',
    0xDEAD,
    3.14,
    TRUE
FROM Demo\Model\User AS u
    INNER JOIN Demo\Model\Post p ON p.userId = u.id
    LEFT OUTER JOIN Demo\Model\Avatar a ON a.userId = u.id
    CROSS JOIN Demo\Model\Tag t
WHERE u.deletedAt IS NULL
GROUP BY u.id, u.name
HAVING COUNT(p.id) > 10
ORDER BY total DESC, u.name ASC
LIMIT 10 OFFSET 20
FOR UPDATE
