SELECT
    CONCAT(u.firstName, ' ', u.lastName) AS fullName,
    LENGTH(u.name),
    LOWER(u.email),
    UPPER(u.country),
    ABS(u.balance),
    SQRT(u.rank),
    MOD(u.id, 10),
    BIT_AND(u.flags, 4),
    SUBSTRING(u.name, 1, 3),
    TRIM(u.name),
    TRIM(BOTH ' ' FROM u.title),
    TRIM(LEADING FROM u.title),
    LOCATE('@', u.email),
    LOCATE('@', u.email, 2),
    SIZE(u.posts),
    IDENTITY(u.address),
    IDENTITY(u.address, 'id'),
    DATE_DIFF(CURRENT_DATE, u.createdAt),
    DATE_ADD(u.createdAt, 7, day),
    DATE_SUB(CURRENT_TIMESTAMP, 1, month),
    CURRENT_TIME,
    COALESCE(u.nickname, u.name, 'anonymous'),
    NULLIF(u.name, ''),
    CASE WHEN u.rank > 100 THEN 'gold' WHEN u.rank > 10 THEN 'silver' ELSE 'bronze' END,
    CASE u.status WHEN 1 THEN 'active' WHEN 2 THEN 'banned' ELSE 'unknown' END
FROM Demo\Entity\User u
